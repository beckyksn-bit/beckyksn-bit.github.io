#include “iic_user.h”

//24c02的相关代码
void iic_24c02_write(unsigned char *pucBuf,unsigned char ucAddr,unsigned char ucNum)
//pucBuf待写入数据的缓冲区首地址 ucAddr：24c02的存储地址 ucNum：要写入的字节数
{
I2CStart();
I2CSendByte(0xa0); //设备地址+读写位
I2CWaitAck();

I2CSendByte(ucAddr);    //存入地址/读取地址
I2CWaitAck();

while(ucNum--)
{
    I2CSendByte(*pucBuff++);
    I2CWaitAck();
}
I2CStop();
delay1(500);//HAL_Delay(5);

}
void iic_24c02_read(unsigned char *pucBuf,unsigned char ucAdrr,unsigned char ucNum)
//pucBuff存储读取内容的首地址 ucAddr:24c02的存储地址 ucNum:要读取的字节数
{
I2C_Start();
I2C_SendByte(0xa0); //传入了24c02的设备地址+写命令
I2C_WaitAck();

I2C_SendByte(ucAddr);   //读取地址（告诉24c02我要读那个位置的数据）
I2C_WaitAck();

I2C_Start();
I2C_SendByte(0xa1)；    //传入了24c02的设备地址+读命令
I2C_WaitAck();

while(ucNum--)
{
    *pucBuf++=I2C_ReceiveByte();
    if(ucNum) 
        I2C_SendAck();
    else 
        I2C_SendNotAck();
}
I2C_Stop();

}

void write_resistor(uint8_t value)
{
I2C_Start();
I2CSendByte(0x5E);
I2CWaitAck();

I2CSendByte(value);
I2CWaitAck();
I2CStop();

}

uint8_t read_resistor(void)
{
uint8_t value;
I2CStart();
I2CSendByte(0x5F);
I2CWaitAck();

value=I2CReceiveByte();
I2CSendNotAck();
I2CStop();

return value;

}

//设置PWM占空比（占空比的单位是%，取值范围0.0-100.0）
void pwm_set_duty(float Duty)
{
TIM2->CCR=(TIM2->ARR+1)*(Duty/100.0f);
}

void pwm_set_frequency(int Frequency)//(一秒钟产生pwm波的个数)
{
uint32_t TIM2_CLK=72000000;
uint32_t ARR_Value;
float duty_ratio=(float)(TIM2->CCR2)/(TIM2->ARR+1);
ARR_Value=(TIM2_CLK/Frequency)-1;
TIM2->ARR=ARR_Value;
TIM2->CCR2=(uint8_t)((ARR_Value+1)*duty_ratio);
TIM2->EGR=TIM_EGR_UG;
}

#include”led_user.h”

void led_disp(uint8_t *ucled)
{
uint8_t temp= 0x00;
static uint8_t temp_old=0xff;

for (uint8_t i=0; i<8; i++)
{
  temp|=ucled[i]<<(7-i);
}
if(temp!=temp_old)
{
GPIOC->ODR&= 0x00ff;
GPIOC->ODR|= ((~temp)&0x00ff)<< 8;
GPIOD->BSRR= 0x01<< 2;
GPIOD->BRR= 0x01<< 2;
temp_old=temp;
}

}

uint8_t ucled[8]={1,1,1,1,0,0,0,0};

void led_proc(){
led_disp(ucled);
}

sheduler_run(){
