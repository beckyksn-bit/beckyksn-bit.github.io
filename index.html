#include “iic_user.h”

//24c02的相关代码
void iic_24c02_write(unsigned char *pucBuf,unsigned char ucAddr,unsigned char ucNum)
//pucBuf待写入数据的缓冲区首地址 ucAddr：24c02的存储地址 ucNum：要写入的字节数
{
I2CStart();
I2CSendByte(0xa0); //设备地址+读写位
I2CWaitAck();

I2CSendByte(ucAddr);    //存入地址/读取地址
I2CWaitAck();

while(ucNum--)
{
    I2CSendByte(*pucBuff++);
    I2CWaitAck();
}
I2CStop();
delay1(500);//HAL_Delay(5);

}
void iic_24c02_read(unsigned char *pucBuf,unsigned char ucAdrr,unsigned char ucNum)
//pucBuff存储读取内容的首地址 ucAddr:24c02的存储地址 ucNum:要读取的字节数
{
I2C_Start();
I2C_SendByte(0xa0); //传入了24c02的设备地址+写命令
I2C_WaitAck();

I2C_SendByte(ucAddr);   //读取地址（告诉24c02我要读那个位置的数据）
I2C_WaitAck();

I2C_Start();
I2C_SendByte(0xa1)；    //传入了24c02的设备地址+读命令
I2C_WaitAck();

while(ucNum--)
{
    *pucBuf++=I2C_ReceiveByte();
    if(ucNum) 
        I2C_SendAck();
    else 
        I2C_SendNotAck();
}
I2C_Stop();

}

void write_resistor(uint8_t value)
{
I2C_Start();
I2CSendByte(0x5E);
I2CWaitAck();

I2CSendByte(value);
I2CWaitAck();
I2CStop();

}

uint8_t read_resistor(void)
{
uint8_t value;
I2CStart();
I2CSendByte(0x5F);
I2CWaitAck();

value=I2CReceiveByte();
I2CSendNotAck();
I2CStop();

return value;

}

//设置PWM占空比（占空比的单位是%，取值范围0.0-100.0）
void pwm_set_duty(float Duty)
{
TIM2->CCR=(TIM2->ARR+1)*(Duty/100.0f);
}

void pwm_set_frequency(int Frequency)//(一秒钟产生pwm波的个数)
{
uint32_t TIM2_CLK=72000000;
uint32_t ARR_Value;
float duty_ratio=(float)(TIM2->CCR2)/(TIM2->ARR+1);
ARR_Value=(TIM2_CLK/Frequency)-1;
TIM2->ARR=ARR_Value;
TIM2->CCR2=(uint8_t)((ARR_Value+1)*duty_ratio);
TIM2->EGR=TIM_EGR_UG;
}

辅助工具理论

串口调试助手模式组合对照表
接收：ASCII 接收数字解析显示为ASCLL码对应字符 eg:发送0x41 串口助手接收为’A’;
发送：ASCII 发送字符解析并发送为ASCLL码值 eg:串口助手发送’A’ 电脑接收到0x41
接收：Hex接收数字解析显示为ASCLL码值 eg:发送’A’ 串口助手接收为0x41;
发送：Hex发送数字解析并发送为对应ASCLL码对应的字符eg:串口助手发送41 电脑接收到’A’;

key_user.h

#include “key_user.h”

uint8_t key_val;
uint8_t key_down;
uint8_t key_up;
uint8_t key_old;

uint8_t key_read()
{
uint8_t temp=0;
if(HAL_GPIO_ReadPin(GPIOB,GPIO_PIN_1)==GPIO_PIN_RESET)
{
temp=1;
}
if(HAL_GPIO_ReadPin(GPIOB,GPIO_PIN_2)==GPIO_PIN_RESET)
{
temp=2;
}
if(HAL_GPIO_ReadPin(GPIOB,GPIO_PIN_3)==GPIO_PIN_RESET)
{
temp=3;
}
if(HAL_GPIO_ReadPin(GPIOB,GPIO_PIN_4)==GPIO_PIN_RESET)
{
temp=4;
}
return temp;
}

void key_proc()
{
key_val=key_read(); //PB2按下瞬间及保持=2，松开=0
key_down=key_val&(key_val^key_old); //2=2&（2^0） 0^x=x;
key_up=(key_val)&(key_val^key_old); //2=0&（0^2）=0xff & 2
key_old=key_val;

if(key_down!=0)
	ucled[key_down-1]^=1;							//对应的led电平翻转

}
